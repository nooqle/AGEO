# 任务：实现前端基础框架

## 背景
基于 Next.js 14 实现前端应用框架，包括路由、状态管理、WebSocket 连接、响应式布局、动画效果等基础设施。

## 参考文档
- `Mittus AEO平台 - 前端交互设计文档 v2.md`

## 技术栈
- Next.js 14 (App Router)
- TypeScript 5
- Tailwind CSS
- Zustand (状态管理)
- socket.io-client (WebSocket)
- Framer Motion (动画)
- Radix UI (基础组件)

```typescript
// frontend/src/stores/conversationStore.ts
import { create } from 'zustand';
import { Message, ConfirmationRequest } from '@/types/message';
import { TPAORUpdate, ExecutionProgress, BrowserState } from '@/types/agent';
import { StopState } from '@/types/execution';

interface ConversationState {
  messages: Message[];
  isAgentTyping: boolean;
  isAgentExecuting: boolean;
  pendingConfirmation: ConfirmationRequest | null;
  
  // TPAOR 状态
  currentTPAOR: {
    thought: string;
    plan: string;
    action: string;
    observation: string;
  };
  
  // 执行进度
  executionProgress: ExecutionProgress | null;
  
  // 浏览器状态（FetchAgent 产生）
  browserState: BrowserState | null;
  
  // 停止状态
  stopState: StopState | null;
  
  // Actions
  addMessage: (message: Message) => void;
  updateMessage: (id: string, updates: Partial<Message>) => void;
  updateTPAOR: (update: TPAORUpdate) => void;
  setExecutionProgress: (progress: ExecutionProgress | null) => void;
  setBrowserState: (state: BrowserState | null) => void;
  setAgentExecuting: (executing: boolean) => void;
  setPendingConfirmation: (request: ConfirmationRequest | null) => void;
  setStopState: (state: StopState | null) => void;
  clearMessagesAfter: (messageId: string) => void;
  reset: () => void;
}

const initialTPAOR = { thought: '', plan: '', action: '', observation: '' };

export const useConversationStore = create<ConversationState>((set, get) => ({
  messages: [],
  isAgentTyping: false,
  isAgentExecuting: false,
  pendingConfirmation: null,
  currentTPAOR: initialTPAOR,
  executionProgress: null,
  browserState: null,
  stopState: null,
  
  addMessage: (message) => set((state) => ({
    messages: [...state.messages, message],
  })),
  
  updateMessage: (id, updates) => set((state) => ({
    messages: state.messages.map((m) =>
      m.id === id ? { ...m, ...updates } : m
    ),
  })),
  
  updateTPAOR: (update) => set((state) => ({
    currentTPAOR: {
      ...state.currentTPAOR,
      [update.phase]: update.content,
    },
  })),
  
  setExecutionProgress: (progress) => set({ executionProgress: progress }),
  
  setBrowserState: (browserState) => set({ browserState }),
  
  setAgentExecuting: (executing) => set({ 
    isAgentExecuting: executing,
    // 开始执行时清空 TPAOR 和停止状态
    ...(executing ? { 
      currentTPAOR: initialTPAOR,
      stopState: null,
    } : {}),
  }),
  
  setPendingConfirmation: (request) => set({ pendingConfirmation: request }),
  
  setStopState: (stopState) => set({ stopState }),
  
  clearMessagesAfter: (messageId) => set((state) => {
    const index = state.messages.findIndex((m) => m.id === messageId);
    if (index === -1) return state;
    return { messages: state.messages.slice(0, index + 1) };
  }),
  
  reset: () => set({
    messages: [],
    isAgentTyping: false,
    isAgentExecuting: false,
    pendingConfirmation: null,
    currentTPAOR: initialTPAOR,
    executionProgress: null,
    browserState: null,
    stopState: null,
  }),
}));


// frontend/src/stores/canvasStore.ts
import { create } from 'zustand';
import { CanvasContent, CanvasMode } from '@/types/canvas';

interface CanvasState {
  isOpen: boolean;
  mode: CanvasMode;
  contents: CanvasContent[];
  activeContentIndex: number;
  
  // Actions
  openCanvas: (content?: CanvasContent) => void;
  closeCanvas: () => void;
  setMode: (mode: CanvasMode) => void;
  addContent: (content: CanvasContent) => void;
  setActiveContent: (index: number) => void;
  setActiveContentById: (id: string) => void;
  removeContent: (id: string) => void;
  removeContentsByIds: (ids: string[]) => void;
  updateContent: (id: string, data: Record<string, any>) => void;
  clearContents: () => void;
}

export const useCanvasStore = create<CanvasState>((set, get) => ({
  isOpen: false,
  mode: 'hidden',
  contents: [],
  activeContentIndex: 0,
  
  openCanvas: (content) => set((state) => {
    const newState: Partial<CanvasState> = {
      isOpen: true,
      mode: 'split',
    };
    if (content) {
      const existingIndex = state.contents.findIndex((c) => c.id === content.id);
      if (existingIndex === -1) {
        newState.contents = [...state.contents, content];
        newState.activeContentIndex = state.contents.length;
      } else {
        newState.activeContentIndex = existingIndex;
      }
    }
    return newState;
  }),
  
  closeCanvas: () => set({ isOpen: false, mode: 'hidden' }),
  
  setMode: (mode) => set({ mode, isOpen: mode !== 'hidden' }),
  
  addContent: (content) => set((state) => ({
    contents: [...state.contents, content],
    activeContentIndex: state.contents.length,
    isOpen: true,
    mode: state.mode === 'hidden' ? 'split' : state.mode,
  })),
  
  setActiveContent: (index) => set({ activeContentIndex: index }),
  
  setActiveContentById: (id) => set((state) => {
    const index = state.contents.findIndex((c) => c.id === id);
    return index !== -1 ? { activeContentIndex: index } : {};
  }),
  
  removeContent: (id) => set((state) => {
    const newContents = state.contents.filter((c) => c.id !== id);
    const newIndex = Math.min(state.activeContentIndex, Math.max(0, newContents.length - 1));
    return {
      contents: newContents,
      activeContentIndex: newIndex,
      isOpen: newContents.length > 0,
      mode: newContents.length > 0 ? state.mode : 'hidden',
    };
  }),
  
  removeContentsByIds: (ids) => set((state) => {
    const idsSet = new Set(ids);
    const newContents = state.contents.filter((c) => !idsSet.has(c.id));
    const newIndex = Math.min(state.activeContentIndex, Math.max(0, newContents.length - 1));
    return {
      contents: newContents,
      activeContentIndex: newIndex,
      isOpen: newContents.length > 0,
      mode: newContents.length > 0 ? state.mode : 'hidden',
    };
  }),
  
  updateContent: (id, data) => set((state) => ({
    contents: state.contents.map((c) =>
      c.id === id ? { ...c, data: { ...c.data, ...data } } : c
    ),
  })),
  
  clearContents: () => set({
    contents: [],
    activeContentIndex: 0,
    isOpen: false,
    mode: 'hidden',
  }),
}));
```

### 5. WebSocket Hook
```typescript
// frontend/src/hooks/useWebSocket.ts
import { useEffect, useRef, useCallback } from 'react';
import { io, Socket } from 'socket.io-client';
import { useConversationStore } from '@/stores/conversationStore';
import { useCanvasStore } from '@/stores/canvasStore';
import { Message } from '@/types/message';

const WS_URL = process.env.NEXT_PUBLIC_WS_URL || 'http://localhost:8000';

export function useWebSocket(sessionId: string | null) {
  const socketRef = useRef<Socket | null>(null);
  
  const {
    addMessage,
    updateTPAOR,
    setExecutionProgress,
    setBrowserState,
    setAgentExecuting,
    setPendingConfirmation,
    setStopState,
  } = useConversationStore();
  
  const { addContent, updateContent } = useCanvasStore();
  
  useEffect(() => {
    if (!sessionId) return;
    
    // 建立连接
    const socket = io(`${WS_URL}`, {
      path: '/ws',
      query: { sessionId },
      transports: ['websocket'],
      reconnection: true,
      reconnectionAttempts: 5,
      reconnectionDelay: 1000,
    });
    
    socketRef.current = socket;
    
    // ========== 事件监听 ==========
    
    socket.on('connect', () => {
      console.log('WebSocket connected');
    });
    
    socket.on('disconnect', (reason) => {
      console.log('WebSocket disconnected:', reason);
    });
    
    // Agent 开始执行
    socket.on('agent_start', () => {
      setAgentExecuting(true);
    });
    
    // TPAOR 状态更新
    socket.on('tpaor_update', (data) => {
      updateTPAOR({
        phase: data.phase,
        content: data.content,
        isComplete: data.is_complete,
      });
    });
    
    // 执行进度更新
    socket.on('execution_progress', (data) => {
      setExecutionProgress({
        stage: data.stage,
        stageName: data.stage_name,
        progress: data.progress,
        status: data.status,
        details: data.details,
        subTasks: data.sub_tasks?.map((t: any) => ({
          name: t.name,
          status: t.status,
        })),
      });
    });
    
    // 浏览器状态更新（FetchAgent 产生）
    socket.on('browser_state', (data) => {
      setBrowserState({
        state: data.state,
        message: data.message,
        platform: data.platform,
        requiresAction: data.requires_action,
        actionHint: data.action_hint,
      });
    });
    
    // 产出内容就绪（触发 Canvas 展开）
    socket.on('output_ready', (data) => {
      addContent({
        id: data.output_id,
        type: data.type,
        title: data.title,
        data: data.full_data,
        createdAt: new Date(),
        relatedMessageId: data.related_message_id || '',
      });
    });
    
    // 确认请求
    socket.on('confirmation_request', (data) => {
      setPendingConfirmation({
        requestId: data.request_id,
        type: data.type,
        message: data.message,
        options: data.options,
        allowTextInput: data.allow_text_input,
        canvasContent: data.canvas_content,
      });
      
      // 如果有 Canvas 内容，添加到 Canvas
      if (data.canvas_content) {
        addContent(data.canvas_content);
      }
    });
    
    // Agent 消息
    socket.on('agent_message', (data) => {
      const message: Message = {
        id: data.id,
        type: 'agent',
        content: data.content,
        timestamp: new Date(data.timestamp),
        tpaor: data.tpaor,
        outputCards: data.output_cards,
        confirmationRequest: data.confirmation_request,
        metadata: data.metadata || {
          canEdit: false,
          canRollback: true,
          relatedOutputIds: data.related_output_ids || [],
        },
      };
      addMessage(message);
    });
    
    // 执行完成
    socket.on('execution_complete', () => {
      setAgentExecuting(false);
      setExecutionProgress(null);
      setBrowserState(null);
      setStopState(null);
    });
    
    // 执行已停止
    socket.on('execution_stopped', (data) => {
      setAgentExecuting(false);
      setExecutionProgress(null);
      setBrowserState(null);
      
      if (data) {
        setStopState({
          isStopped: true,
          completedStages: data.completed_stages || [],
          pendingStages: data.pending_stages || [],
          partialResults: data.partial_results,
        });
      }
    });
    
    // 错误
    socket.on('error', (data) => {
      console.error('WebSocket error:', data);
      setAgentExecuting(false);
    });
    
    // 心跳
    const heartbeat = setInterval(() => {
      if (socket.connected) {
        socket.emit('ping');
      }
    }, 30000);
    
    socket.on('pong', () => {
      // 心跳响应
    });
    
    return () => {
      clearInterval(heartbeat);
      socket.disconnect();
      socketRef.current = null;
    };
  }, [sessionId]);
  
  // ========== 发送方法 ==========
  
  // 发送用户消息
  const sendMessage = useCallback((content: string) => {
    if (!socketRef.current?.connected) return;
    
    socketRef.current.emit('user_message', { content });
    
    // 立即添加用户消息到本地
    const message: Message = {
      id: `temp-${Date.now()}`,
      type: 'user',
      content,
      timestamp: new Date(),
      metadata: { canEdit: true, canRollback: true, relatedOutputIds: [] },
    };
    addMessage(message);
  }, [addMessage]);
  
  // 发送确认
  const sendConfirmation = useCallback((requestId: string, selection: any) => {
    if (!socketRef.current?.connected) return;
    
    socketRef.current.emit('confirmation', {
      request_id: requestId,
      selection,
    });
    setPendingConfirmation(null);
  }, [setPendingConfirmation]);
  
  // 停止执行
  const stopExecution = useCallback(() => {
    if (!socketRef.current?.connected) return;
    socketRef.current.emit('stop');
  }, []);
  
  // 继续执行
  const resumeExecution = useCallback(() => {
    if (!socketRef.current?.connected) return;
    socketRef.current.emit('resume');
    setStopState(null);
  }, [setStopState]);
  
  // 回退
  const rollback = useCallback((messageId: string) => {
    if (!socketRef.current?.connected) return;
    socketRef.current.emit('rollback', { message_id: messageId });
  }, []);
  
  return {
    sendMessage,
    sendConfirmation,
    stopExecution,
    resumeExecution,
    rollback,
    isConnected: socketRef.current?.connected ?? false,
  };
}
```

```typescript
// frontend/src/hooks/useResponsive.ts
import { useState, useEffect } from 'react';

type Breakpoint = 'mobile' | 'tablet' | 'laptop' | 'desktop';

interface ResponsiveState {
  breakpoint: Breakpoint;
  width: number;
  isMobile: boolean;
  isTablet: boolean;
  isLaptop: boolean;
  isDesktop: boolean;
  canShowSplitCanvas: boolean;
  canShowSideBySide: boolean;
}

const BREAKPOINTS = {
  mobile: 0,
  tablet: 768,
  laptop: 1024,
  desktop: 1440,
};

export function useResponsive(): ResponsiveState {
  const [state, setState] = useState<ResponsiveState>({
    breakpoint: 'desktop',
    width: typeof window !== 'undefined' ? window.innerWidth : 1440,
    isMobile: false,
    isTablet: false,
    isLaptop: false,
    isDesktop: true,
    canShowSplitCanvas: true,
    canShowSideBySide: true,
  });
  
  useEffect(() => {
    const handleResize = () => {
      const width = window.innerWidth;
      let breakpoint: Breakpoint = 'desktop';
      
      if (width < BREAKPOINTS.tablet) {
        breakpoint = 'mobile';
      } else if (width < BREAKPOINTS.laptop) {
        breakpoint = 'tablet';
      } else if (width < BREAKPOINTS.desktop) {
        breakpoint = 'laptop';
      }
      
      setState({
        breakpoint,
        width,
        isMobile: breakpoint === 'mobile',
        isTablet: breakpoint === 'tablet',
        isLaptop: breakpoint === 'laptop',
        isDesktop: breakpoint === 'desktop',
        canShowSplitCanvas: breakpoint === 'laptop' || breakpoint === 'desktop',
        canShowSideBySide: breakpoint === 'desktop',
      });
    };
    
    handleResize();
    window.addEventListener('resize', handleResize);
    return () => window.removeEventListener('resize', handleResize);
  }, []);
  
  return state;
}
```

```typescript
// frontend/src/hooks/useKeyboardShortcuts.ts
import { useEffect, useCallback } from 'react';
import { useConversationStore } from '@/stores/conversationStore';
import { useCanvasStore } from '@/stores/canvasStore';

interface KeyboardShortcutsOptions {
  onStopExecution?: () => void;
  onToggleCanvas?: () => void;
}

export function useKeyboardShortcuts(options: KeyboardShortcutsOptions = {}) {
  const { onStopExecution, onToggleCanvas } = options;
  const { isAgentExecuting } = useConversationStore();
  const { isOpen, closeCanvas, setMode } = useCanvasStore();
  
  const handleKeyDown = useCallback((e: KeyboardEvent) => {
    // 如果在输入框中，忽略部分快捷键
    const isInputFocused = ['INPUT', 'TEXTAREA'].includes(
      (e.target as HTMLElement).tagName
    );
    
    // Esc - 停止执行或关闭 Canvas
    if (e.key === 'Escape') {
      e.preventDefault();
      if (isAgentExecuting && onStopExecution) {
        onStopExecution();
      } else if (isOpen) {
        closeCanvas();
      }
      return;
    }
    
    // 如果在输入框中，以下快捷键不生效
    if (isInputFocused) return;
    
    // Cmd/Ctrl + \ - 切换 Canvas
    if ((e.metaKey || e.ctrlKey) && e.key === '\\') {
      e.preventDefault();
      if (onToggleCanvas) {
        onToggleCanvas();
      } else if (isOpen) {
        closeCanvas();
      }
      return;
    }
    
    // Cmd/Ctrl + Shift + F - 聚焦模式
    if ((e.metaKey || e.ctrlKey) && e.shiftKey && e.key === 'F') {
      e.preventDefault();
      if (isOpen) {
        setMode('focused');
      }
      return;
    }
  }, [isAgentExecuting, isOpen, onStopExecution, onToggleCanvas, closeCanvas, setMode]);
  
  useEffect(() => {
    window.addEventListener('keydown', handleKeyDown);
    return () => window.removeEventListener('keydown', handleKeyDown);
  }, [handleKeyDown]);
}
```

```typescript
// 解析用户选择输入
export function parseUserSelection(input: string): string[] | null {
  // 支持格式: "①②", "1,2", "1和2", "选1和2"
  const matches = input.match(/[①②③④⑤⑥⑦⑧⑨⑩\d]/g);
  if (!matches) return null;
  
  const numberMap: Record<string, string> = {
    '①': '1', '②': '2', '③': '3', '④': '4', '⑤': '5',
    '⑥': '6', '⑦': '7', '⑧': '8', '⑨': '9', '⑩': '10',
  };
  
  return [...new Set(matches.map((m) => numberMap[m] || m))];
}

// 格式化选择为消息
export function formatSelectionAsMessage(
  selections: { id: string; label: string }[]
): string {
  const labels = selections.map((s, i) => `${getCircledNumber(i + 1)} ${s.label}`);
  return `已选择：${labels.join('、')}`;
}

// 获取带圈数字
export function getCircledNumber(n: number): string {
  const circled = ['①', '②', '③', '④', '⑤', '⑥', '⑦', '⑧', '⑨', '⑩'];
  return circled[n - 1] || String(n);
}

// 截断文本
export function truncate(text: string, maxLength: number): string {
  if (text.length <= maxLength) return text;
  return text.slice(0, maxLength - 3) + '...';
}
```

```typescript
// frontend/src/services/api.ts
const API_URL = process.env.NEXT_PUBLIC_API_URL || 'http://localhost:8000/api/v1';

class ApiService {
  private async request<T>(
    endpoint: string,
    options: RequestInit = {}
  ): Promise<T> {
    const url = `${API_URL}${endpoint}`;
    const response = await fetch(url, {
      ...options,
      headers: {
        'Content-Type': 'application/json',
        ...options.headers,
      },
    });
    
    if (!response.ok) {
      const error = await response.json().catch(() => ({}));
      throw new Error(error.detail || `Request failed: ${response.status}`);
    }
    
    return response.json();
  }
  
  // 会话管理
  async createSession() {
    return this.request<{ id: string; created_at: string }>('/sessions', {
      method: 'POST',
    });
  }
  
  async getSession(sessionId: string) {
    return this.request<any>(`/sessions/${sessionId}`);
  }
  
  async deleteSession(sessionId: string) {
    return this.request<void>(`/sessions/${sessionId}`, {
      method: 'DELETE',
    });
  }
  
  // 消息管理
  async getMessages(sessionId: string, options?: { limit?: number; before?: string }) {
    const params = new URLSearchParams();
    if (options?.limit) params.set('limit', String(options.limit));
    if (options?.before) params.set('before', options.before);
    
    const query = params.toString();
    return this.request<any[]>(`/sessions/${sessionId}/messages${query ? `?${query}` : ''}`);
  }
  
  async sendMessage(sessionId: string, content: string) {
    return this.request<any>(`/sessions/${sessionId}/messages`, {
      method: 'POST',
      body: JSON.stringify({ content }),
    });
  }
  
  async rollbackAfterMessage(sessionId: string, messageId: string) {
    return this.request<any>(`/sessions/${sessionId}/messages/${messageId}/after`, {
      method: 'DELETE',
    });
  }
  
  // 产出管理
  async getOutputs(sessionId: string) {
    return this.request<any[]>(`/sessions/${sessionId}/outputs`);
  }
  
  async getOutput(sessionId: string, outputId: string) {
    return this.request<any>(`/sessions/${sessionId}/outputs/${outputId}`);
  }
  
  async exportOutput(sessionId: string, outputId: string, format: 'pdf' | 'excel') {
    const response = await fetch(
      `${API_URL}/sessions/${sessionId}/outputs/${outputId}/export?format=${format}`
    );
    if (!response.ok) throw new Error('Export failed');
    return response.blob();
  }
  
  // Agent 控制
  async stopAgent(sessionId: string) {
    return this.request<any>(`/sessions/${sessionId}/agent/stop`, {
      method: 'POST',
    });
  }
  
  async resumeAgent(sessionId: string) {
    return this.request<any>(`/sessions/${sessionId}/agent/resume`, {
      method: 'POST',
    });
  }
  
  async sendConfirmation(sessionId: string, requestId: string, selection: any) {
    return this.request<any>(`/sessions/${sessionId}/agent/confirm`, {
      method: 'POST',
      body: JSON.stringify({ request_id: requestId, selection }),
    });
  }
}

export const api = new ApiService();
```
